//===== rAthena Script ========================================
//= AI Companion
//= Hire up to 3 AI soldiers (slave mobs) that follow the player
//= and fight monsters alongside them.
//= Uses the native 'summon' command, no C++ changes needed.
//= Each companion is stored in an array slot (0..2).
//= Tune stat formulas inside OnScale / OnScaleFull.
//=============================================================

prontera,156,193,5	script	AI Companion	123,{
	mes "[AI Companion]";
	mes "Greetings, adventurer.";
	mes "I can summon up to 3 AI";
	mes "soldiers to be your";
	mes "companions. They will";
	mes "follow you and fight by";
	mes "your side, sharing the";
	mes "adventure with you!";
	next;
	switch(select("Hire a companion:Release companion:Check companions:Equip your companion")) {
	case 1:
		callsub OnHireMenu;
		close;
	case 2:
		callsub OnRelease;
		close;
	case 3:
		callsub OnStatus;
		close;
	case 4:
		callsub OnEquipMenu;
		close;
	}

OnHireMenu:
	set .@slots, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1)
		if (#COMPANION_CLASS[.@i])
			set .@slots, .@slots + 1;
	if (.@slots >= 3) {
		mes "[AI Companion]";
		mes "Your companion party is";
		mes "full (3/3). Release one";
		mes "first if you want to";
		mes "hire a different one.";
		close;
	}
	mes "[AI Companion]";
	mes "You have " + .@slots + "/3 companions.";
	mes "Choose your next AI friend:";
	next;
	switch(select("Seyren - Lord Knight (tank):Eremes - Assassin (crit):Cenia - Priestess (magic):Howard - Creator (atk):Margaretha - High Priest (support):Cecil - Gunslinger (dex):Kathryne - High Wizard (nuke):Randel - Crusader (tank):Flamel - Paladin (holy tank):Celia - Wizard (fire):Chen - Monk (dmg):Gertie - Rogue (evade)")) {
	case 1:
		set .@cls, 1799;
		set .@nm$, "Seyren";
		break;
	case 2:
		set .@cls, 1800;
		set .@nm$, "Eremes";
		break;
	case 3:
		set .@cls, 1658;
		set .@nm$, "Cenia";
		break;
	case 4:
		set .@cls, 1636;
		set .@nm$, "Howard";
		break;
	case 5:
		set .@cls, 1637;
		set .@nm$, "Margaretha";
		break;
	case 6:
		set .@cls, 1638;
		set .@nm$, "Cecil";
		break;
	case 7:
		set .@cls, 1639;
		set .@nm$, "Kathryne";
		break;
	case 8:
		set .@cls, 3226;
		set .@nm$, "Randel";
		break;
	case 9:
		set .@cls, 3227;
		set .@nm$, "Flamel";
		break;
	case 10:
		set .@cls, 3228;
		set .@nm$, "Celia";
		break;
	case 11:
		set .@cls, 3229;
		set .@nm$, "Chen";
		break;
	case 12:
		set .@cls, 3230;
		set .@nm$, "Gertie";
		break;
	}
	set .@idx, -1;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (!#COMPANION_CLASS[.@i]) {
			set .@idx, .@i;
			break;
		}
	}
	set #COMPANION_CLASS[.@idx], .@cls;
	set #COMPANION_NAME$[.@idx], .@nm$;
	set #COMPANION_GID[.@idx], 0;
	set #COMPANION_DEAD_TICK[.@idx], 0;
	if (!#COMPANION_LEVEL[.@cls])
		set #COMPANION_LEVEL[.@cls], 1;
	if (!#COMPANION_EXP[.@cls])
		set #COMPANION_EXP[.@cls], 0;
	callsub OnSummon, .@idx;
	if (!#COMPANION_GID[.@idx]) {
		set #COMPANION_CLASS[.@idx], 0;
		set #COMPANION_NAME$[.@idx], "";
		mes "[AI Companion]";
		mes "Failed to summon your friend...";
		mes "Please try again.";
		close;
	}
	mes "[AI Companion]";
	mes "Your AI companion " + #COMPANION_NAME$[.@idx];
	mes "has been summoned. It will";
	mes "follow you and fight by";
	mes "your side!";
	addtimer 5000, "AI Companion::OnCompanionTick";
	close;

OnRelease:
	callsub OnPickSlot;
	if (@pick_slot < 0)
		close;
	set .@idx, @pick_slot;
	if (#COMPANION_GID[.@idx])
		unitkill #COMPANION_GID[.@idx];
	set #COMPANION_CLASS[.@idx], 0;
	set #COMPANION_GID[.@idx], 0;
	set #COMPANION_DEAD_TICK[.@idx], 0;
	set #COMPANION_NAME$[.@idx], "";
	if (#COMPANION_WEAPON[.@idx]) {
		getitem #COMPANION_WEAPON[.@idx], 1;
		set #COMPANION_WEAPON[.@idx], 0;
	}
	if (#COMPANION_SHIELD[.@idx]) {
		getitem #COMPANION_SHIELD[.@idx], 1;
		set #COMPANION_SHIELD[.@idx], 0;
	}
	if (#COMPANION_HELM[.@idx]) {
		getitem #COMPANION_HELM[.@idx], 1;
		set #COMPANION_HELM[.@idx], 0;
	}
	if (#COMPANION_ARMOR[.@idx]) {
		getitem #COMPANION_ARMOR[.@idx], 1;
		set #COMPANION_ARMOR[.@idx], 0;
	}
	atcommand "@save";
	mes "[AI Companion]";
	mes "Your companion has been released.";
	mes "Farewell, and stay safe!";
	close;

OnStatus:
	set .@cnt, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set .@cnt, .@cnt + 1;
			mes "[AI Companion]";
			mes "Slot " + (.@i + 1) + " : " + #COMPANION_NAME$[.@i];
			mes "Level     : " + #COMPANION_LEVEL[#COMPANION_CLASS[.@i]] + " / 999";
			callsub OnNextExp, #COMPANION_LEVEL[#COMPANION_CLASS[.@i]];
			mes "EXP       : " + #COMPANION_EXP[#COMPANION_CLASS[.@i]] + " / " + @next_exp;
			if (#COMPANION_GID[.@i] && unitexists(#COMPANION_GID[.@i])) {
				set .@d[UMOB_MASTERAID], -1;
				getunitdata(#COMPANION_GID[.@i], .@d);
				if (.@d[UMOB_MASTERAID] == getcharid(3))
					mes "Status    : Alive";
				else
					mes "Status    : Recovering...";
			} else
				mes "Status    : Recovering...";
			next;
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have a companion.";
		close;
	}
	close;

OnPickSlot:
	set .@menu$, "";
	set .@cnt, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set .@cnt, .@cnt + 1;
			set .@slot[.@cnt], .@i;
			set .@menu$, .@menu$ + #COMPANION_NAME$[.@i];
			if (.@cnt == 1)
				set .@menu$, .@menu$ + " (Lv " + #COMPANION_LEVEL[#COMPANION_CLASS[.@i]] + ")";
			set .@menu$, .@menu$ + ":";
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have a companion.";
		set @pick_slot, -1;
		return;
	}
	if (.@cnt == 1) {
		set @pick_slot, .@slot[1];
		return;
	}
	mes "[AI Companion]";
	mes "Choose a companion:";
	next;
	set .@sel, select(.@menu$ + "Cancel");
	if (.@sel == .@cnt + 1) {
		set @pick_slot, -1;
		return;
	}
	set @pick_slot, .@slot[.@sel];
	return;

OnSummon:
	set .@idx, getarg(0);
	set @gid, summon(#COMPANION_NAME$[.@idx], #COMPANION_CLASS[.@idx], 2147483647, "AI Companion::OnCompanionDead");
	set #COMPANION_GID[.@idx], @gid;
	if (!@gid)
		return;
	callsub OnScaleFull, .@idx;
	return;

OnScale:
	set .@idx, getarg(0);
	if (#COMPANION_LEVEL[#COMPANION_CLASS[.@idx]] < 1)
		set #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]], 1;
	set @gid, #COMPANION_GID[.@idx];
	if (!@gid || !unitexists(@gid))
		return;
	set @lvl, #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]];
	setunitdata(@gid, UMOB_LEVEL, @lvl);
	setunitdata(@gid, UMOB_STR, 50 + @lvl * 2);
	setunitdata(@gid, UMOB_AGI, 40 + @lvl * 2);
	setunitdata(@gid, UMOB_VIT, 40 + @lvl * 2);
	setunitdata(@gid, UMOB_INT, 30 + @lvl * 2);
	setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
	setunitdata(@gid, UMOB_LUK, 25 + @lvl);
	setunitdata(@gid, UMOB_ATKMIN, @lvl * 20);
	setunitdata(@gid, UMOB_ATKMAX, @lvl * 28);
	setunitdata(@gid, UMOB_DEF, @lvl / 2 + 10);
	setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 10);
	setunitdata(@gid, UMOB_HIT, 100 + @lvl);
	setunitdata(@gid, UMOB_FLEE, 80 + @lvl);
	setunitdata(@gid, UMOB_CRIT, 20);
	if (#COMPANION_CLASS[.@idx] == 1799) {
		setunitdata(@gid, UMOB_VIT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_STR, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 1800) {
		setunitdata(@gid, UMOB_AGI, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_LUK, 40 + @lvl * 2);
		setunitdata(@gid, UMOB_CRIT, 40);
		setunitdata(@gid, UMOB_FLEE, 100 + @lvl);
	} else if (#COMPANION_CLASS[.@idx] == 1658 || #COMPANION_CLASS[.@idx] == 3228) {
		setunitdata(@gid, UMOB_INT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 18);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 26);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 16);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 22);
	} else if (#COMPANION_CLASS[.@idx] == 1636) {
		setunitdata(@gid, UMOB_STR, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 24);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 1637) {
		setunitdata(@gid, UMOB_INT, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_VIT, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 12);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 18);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 1638) {
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_AGI, 50 + @lvl * 2);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 22);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 30);
	} else if (#COMPANION_CLASS[.@idx] == 1639) {
		setunitdata(@gid, UMOB_INT, 70 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 22);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 3226) {
		setunitdata(@gid, UMOB_VIT, 55 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
	} else if (#COMPANION_CLASS[.@idx] == 3227) {
		setunitdata(@gid, UMOB_VIT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_STR, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 3229) {
		setunitdata(@gid, UMOB_STR, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_AGI, 50 + @lvl * 2);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 24);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 3230) {
		setunitdata(@gid, UMOB_AGI, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_FLEE, 110 + @lvl);
		setunitdata(@gid, UMOB_CRIT, 30);
	} else {
		setunitdata(@gid, UMOB_INT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 18);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 26);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 16);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 22);
	}
	if (#COMPANION_BS[.@idx * 20 + 1])
		setunitdata(@gid, UMOB_STR, #COMPANION_BS[.@idx * 20 + 1] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 1]));
	if (#COMPANION_BS[.@idx * 20 + 2])
		setunitdata(@gid, UMOB_AGI, #COMPANION_BS[.@idx * 20 + 2] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 2]));
	if (#COMPANION_BS[.@idx * 20 + 3])
		setunitdata(@gid, UMOB_VIT, #COMPANION_BS[.@idx * 20 + 3] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 3]));
	if (#COMPANION_BS[.@idx * 20 + 4])
		setunitdata(@gid, UMOB_INT, #COMPANION_BS[.@idx * 20 + 4] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 4]));
	if (#COMPANION_BS[.@idx * 20 + 5])
		setunitdata(@gid, UMOB_DEX, #COMPANION_BS[.@idx * 20 + 5] + 3 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 5]));
	if (#COMPANION_BS[.@idx * 20 + 6])
		setunitdata(@gid, UMOB_LUK, #COMPANION_BS[.@idx * 20 + 6] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 6]));
	if (#COMPANION_BS[.@idx * 20 + 7]) {
		setunitdata(@gid, UMOB_ATKMIN, #COMPANION_BS[.@idx * 20 + 7] + 24 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 7]));
		setunitdata(@gid, UMOB_ATKMAX, (#COMPANION_BS[.@idx * 20 + 7] + 24 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 7])) * 4 / 3);
	}
	if (#COMPANION_BS[.@idx * 20 + 8]) {
		setunitdata(@gid, UMOB_MATKMIN, #COMPANION_BS[.@idx * 20 + 8] + 20 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 8]));
		setunitdata(@gid, UMOB_MATKMAX, (#COMPANION_BS[.@idx * 20 + 8] + 20 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 8])) * 3 / 2);
	}
	if (#COMPANION_BS[.@idx * 20 + 10])
		setunitdata(@gid, UMOB_DEF, #COMPANION_BS[.@idx * 20 + 10] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 10]));
	if (#COMPANION_BS[.@idx * 20 + 11])
		setunitdata(@gid, UMOB_MDEF, #COMPANION_BS[.@idx * 20 + 11] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 11]));
	if (#COMPANION_BS[.@idx * 20 + 16])
		setunitdata(@gid, UMOB_PDODGE, #COMPANION_BS[.@idx * 20 + 16] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 16]));
	if (#COMPANION_BS[.@idx * 20 + 17])
		setunitdata(@gid, UMOB_RES, #COMPANION_BS[.@idx * 20 + 17] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 17]));
	if (#COMPANION_BS[.@idx * 20 + 18])
		setunitdata(@gid, UMOB_MRES, #COMPANION_BS[.@idx * 20 + 18] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 18]));
	set .@d[UMOB_ATKMIN], 0;
	set .@d[UMOB_ATKMAX], 0;
	set .@d[UMOB_DEF], 0;
	set .@d[UMOB_MDEF], 0;
	set .@d[UMOB_MATKMIN], 0;
	set .@d[UMOB_MATKMAX], 0;
	getunitdata(@gid, .@d);
	set @ba, 0;
	set @bm, 0;
	set @bd, 0;
	set @bmd, 0;
	if (#COMPANION_WEAPON[.@idx]) {
		set @ba, getiteminfo(#COMPANION_WEAPON[.@idx], ITEMINFO_ATTACK);
		set @bm, getiteminfo(#COMPANION_WEAPON[.@idx], ITEMINFO_MAGICATTACK);
	}
	if (#COMPANION_SHIELD[.@idx]) {
		set @bd, @bd + getiteminfo(#COMPANION_SHIELD[.@idx], ITEMINFO_DEFENSE) / 2;
		set @bmd, @bmd + getiteminfo(#COMPANION_SHIELD[.@idx], ITEMINFO_DEFENSE) / 4;
	}
	if (#COMPANION_HELM[.@idx])
		set @bd, @bd + getiteminfo(#COMPANION_HELM[.@idx], ITEMINFO_DEFENSE) / 3;
	if (#COMPANION_ARMOR[.@idx]) {
		set @bd, @bd + getiteminfo(#COMPANION_ARMOR[.@idx], ITEMINFO_DEFENSE);
		set @bmd, @bmd + getiteminfo(#COMPANION_ARMOR[.@idx], ITEMINFO_DEFENSE) / 2;
	}
	if (@ba) {
		setunitdata(@gid, UMOB_ATKMIN, .@d[UMOB_ATKMIN] + @ba);
		setunitdata(@gid, UMOB_ATKMAX, .@d[UMOB_ATKMAX] + @ba);
	}
	if (@bm) {
		setunitdata(@gid, UMOB_MATKMIN, .@d[UMOB_MATKMIN] + @bm);
		setunitdata(@gid, UMOB_MATKMAX, .@d[UMOB_MATKMAX] + @bm);
	}
	if (@bd)
		setunitdata(@gid, UMOB_DEF, .@d[UMOB_DEF] + @bd);
	if (@bmd)
		setunitdata(@gid, UMOB_MDEF, .@d[UMOB_MDEF] + @bmd);
	if (#COMPANION_BS[.@idx * 20 + 12])
		setunitdata(@gid, UMOB_HIT, #COMPANION_BS[.@idx * 20 + 12] + 3 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 12]));
	if (#COMPANION_BS[.@idx * 20 + 13])
		setunitdata(@gid, UMOB_FLEE, #COMPANION_BS[.@idx * 20 + 13] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 13]));
	if (#COMPANION_BS[.@idx * 20 + 14])
		setunitdata(@gid, UMOB_CRIT, #COMPANION_BS[.@idx * 20 + 14] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 14]));
	return;

OnScaleFull:
	set .@idx, getarg(0);
	set @gid, #COMPANION_GID[.@idx];
	if (!@gid || !unitexists(@gid))
		return;
	callsub OnScale, .@idx;
	set @lvl, #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]];
	setunitdata(@gid, UMOB_CLASS, #COMPANION_CLASS[.@idx]);
	setunitdata(@gid, UMOB_MODE, MD_CANMOVE|MD_CANATTACK|MD_AGGRESSIVE|MD_ASSIST);
	setunitdata(@gid, UMOB_AI, AI_ATTACK);
	setunitdata(@gid, UMOB_RACE, RC_DEMIHUMAN);
	setunitdata(@gid, UMOB_ELETYPE, ELE_NEUTRAL);
	setunitdata(@gid, UMOB_ELELEVEL, 1);
	setunitdata(@gid, UMOB_SPEED, 100);
	setunitdata(@gid, UMOB_ATKRANGE, 1);
	setunitdata(@gid, UMOB_AMOTION, 480);
	setunitdata(@gid, UMOB_ADELAY, 480);
	if (#COMPANION_BS[.@idx * 20 + 15]) {
		set .@am, 2000 - #COMPANION_BS[.@idx * 20 + 15] * 10;
		if (.@am < 1) set .@am, 1;
		setunitdata(@gid, UMOB_AMOTION, .@am);
		setunitdata(@gid, UMOB_ADELAY, .@am);
	}
	setunitdata(@gid, UMOB_WEAPON, #COMPANION_WEAPON[.@idx] ? getiteminfo(#COMPANION_WEAPON[.@idx], ITEMINFO_VIEW) : 0);
	setunitdata(@gid, UMOB_SHIELD, #COMPANION_SHIELD[.@idx] ? getiteminfo(#COMPANION_SHIELD[.@idx], ITEMINFO_VIEW) : 0);
	setunitdata(@gid, UMOB_HEADTOP, #COMPANION_HELM[.@idx] ? getiteminfo(#COMPANION_HELM[.@idx], ITEMINFO_VIEW) : 0);
	if (#COMPANION_CLASS[.@idx] == 1799 || #COMPANION_CLASS[.@idx] == 3226 || #COMPANION_CLASS[.@idx] == 3227) {
		setunitdata(@gid, UMOB_MAXHP, 800 + @lvl * 320);
		setunitdata(@gid, UMOB_HP, 800 + @lvl * 320);
	} else if (#COMPANION_CLASS[.@idx] == 1800 || #COMPANION_CLASS[.@idx] == 3229 || #COMPANION_CLASS[.@idx] == 3230) {
		setunitdata(@gid, UMOB_MAXHP, 650 + @lvl * 260);
		setunitdata(@gid, UMOB_HP, 650 + @lvl * 260);
	} else if (#COMPANION_CLASS[.@idx] == 1638) {
		setunitdata(@gid, UMOB_MAXHP, 600 + @lvl * 250);
		setunitdata(@gid, UMOB_HP, 600 + @lvl * 250);
	} else {
		setunitdata(@gid, UMOB_MAXHP, 550 + @lvl * 240);
		setunitdata(@gid, UMOB_HP, 550 + @lvl * 240);
	}
	if (#COMPANION_BS[.@idx * 20 + 9]) {
		setunitdata(@gid, UMOB_MAXHP, #COMPANION_BS[.@idx * 20 + 9] + 260 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 9]));
		setunitdata(@gid, UMOB_HP, #COMPANION_BS[.@idx * 20 + 9] + 260 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 9]));
	}
	if (#COMPANION_BS[.@idx * 20 + 12])
		setunitdata(@gid, UMOB_HIT, #COMPANION_BS[.@idx * 20 + 12] + 3 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 12]));
	if (#COMPANION_BS[.@idx * 20 + 13])
		setunitdata(@gid, UMOB_FLEE, #COMPANION_BS[.@idx * 20 + 13] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 13]));
	if (#COMPANION_BS[.@idx * 20 + 14])
		setunitdata(@gid, UMOB_CRIT, #COMPANION_BS[.@idx * 20 + 14] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 14]));
	return;

OnCompanionTick:
	set @has_any, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set @has_any, 1;
			if (#COMPANION_GID[.@i] && unitexists(#COMPANION_GID[.@i])) {
				set .@d[UMOB_MASTERAID], -1;
				set .@d[UMOB_MAPID], -1;
				set .@d[UMOB_X], -1;
				set .@d[UMOB_Y], -1;
				getunitdata(#COMPANION_GID[.@i], .@d);
				if (.@d[UMOB_MASTERAID] == getcharid(3) && .@d[UMOB_MAPID] >= 0) {
					callsub OnScale, .@i;
					callsub OnCompanionBuff, #COMPANION_CLASS[.@i], #COMPANION_GID[.@i];
					getmapxy(@comp_map$, @comp_x, @comp_y, BL_PC);
					if (@comp_x != -1 && @comp_y != -1) {
						set @df, distance(@comp_x, @comp_y, .@d[UMOB_X], .@d[UMOB_Y]);
						if (.@d[UMOB_MAPID] != mapname2id(strcharinfo(3)) || @df > 15)
							unitwarp(#COMPANION_GID[.@i], @comp_map$, @comp_x, @comp_y);
					}
				} else {
					set #COMPANION_GID[.@i], 0;
					if (#COMPANION_DEAD_TICK[.@i] == 0) {
						set #COMPANION_DEAD_TICK[.@i], gettimetick(2);
						dispbottom #COMPANION_NAME$[.@i] + " has fallen...";
					}
					if (gettimetick(2) - #COMPANION_DEAD_TICK[.@i] < 10) {
						// wait for revive cooldown
					} else {
						callsub OnSummon, .@i;
						if (#COMPANION_GID[.@i]) {
							set #COMPANION_DEAD_TICK[.@i], 0;
							dispbottom #COMPANION_NAME$[.@i] + " has returned!";
						}
					}
				}
			} else {
				if (#COMPANION_DEAD_TICK[.@i] == 0) {
					set #COMPANION_DEAD_TICK[.@i], gettimetick(2);
					dispbottom #COMPANION_NAME$[.@i] + " has fallen...";
				}
				if (gettimetick(2) - #COMPANION_DEAD_TICK[.@i] < 10) {
					// wait for revive cooldown
				} else {
					callsub OnSummon, .@i;
					if (#COMPANION_GID[.@i]) {
						set #COMPANION_DEAD_TICK[.@i], 0;
						dispbottom #COMPANION_NAME$[.@i] + " has returned!";
					}
				}
			}
		}
	}
	if (@has_any)
		addtimer 5000, "AI Companion::OnCompanionTick";
	end;

OnNPCKillEvent:
	set @comp_killed, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (!#COMPANION_CLASS[.@i])
			continue;
		if (#COMPANION_LEVEL[#COMPANION_CLASS[.@i]] >= 999)
			continue;
		if (!#COMPANION_GID[.@i])
			continue;
		if (#COMPANION_GID[.@i] != killerrid)
			continue;
		if (!unitexists(#COMPANION_GID[.@i]))
			continue;
		set .@d[UMOB_MASTERAID], -1;
		set .@d[UMOB_X], -1;
		set .@d[UMOB_Y], -1;
		getunitdata(#COMPANION_GID[.@i], .@d);
		if (.@d[UMOB_MASTERAID] != getcharid(3))
			continue;
		set @comp_killed, 1;
		set @kill_x, .@d[UMOB_X];
		set @kill_y, .@d[UMOB_Y];
		set @moblv, getmonsterinfo(killedrid, MOB_LV);
		if (@moblv < 1)
			set @moblv, 1;
		set #COMPANION_EXP[#COMPANION_CLASS[.@i]], #COMPANION_EXP[#COMPANION_CLASS[.@i]] + @moblv * 15 + 50;
		callsub OnNextExp, #COMPANION_LEVEL[#COMPANION_CLASS[.@i]];
		while (#COMPANION_LEVEL[#COMPANION_CLASS[.@i]] < 999 && #COMPANION_EXP[#COMPANION_CLASS[.@i]] >= @next_exp) {
			set #COMPANION_EXP[#COMPANION_CLASS[.@i]], #COMPANION_EXP[#COMPANION_CLASS[.@i]] - @next_exp;
			set #COMPANION_LEVEL[#COMPANION_CLASS[.@i]], #COMPANION_LEVEL[#COMPANION_CLASS[.@i]] + 1;
			callsub OnLevelUp, .@i;
			callsub OnNextExp, #COMPANION_LEVEL[#COMPANION_CLASS[.@i]];
		}
		break;
	}
	if (@comp_killed)
		callsub OnVacuumLoot, @kill_x, @kill_y;
	end;

OnVacuumLoot:
	set .@kx, getarg(0);
	set .@ky, getarg(1);
	getmapxy(@v_map$, @v_px, @v_py, BL_PC);
	if (@v_px == -1)
		return;
	set .@num, getmapunits(BL_ITEM, @v_map$, .@ig[0]);
	for (set .@k, 0; .@k < .@num; set .@k, .@k + 1) {
		set .@d[UMOB_X], -1;
		set .@d[UMOB_Y], -1;
		getunitdata(.@ig[.@k], .@d);
		if (.@d[UMOB_X] == -1 || .@d[UMOB_Y] == -1)
			continue;
		set .@dd, distance(.@d[UMOB_X], .@d[UMOB_Y], .@kx, .@ky);
		if (.@dd <= 5)
			unitwarp(.@ig[.@k], @v_map$, @v_px, @v_py);
	}
	return;

OnNextExp:
	set .@lvl, getarg(0);
	if (.@lvl <= BaseLevel)
		set @next_exp, getbaseexp_ratio(100, .@lvl);
	else
		set @next_exp, .@lvl * .@lvl * 5 + .@lvl * 100;
	return;

	callsub OnScaleFull, .@idx;
	return;

OnEquipSlot:
	set .@idx, getarg(0);
	set .@slot, getarg(1);
	getinventorylist();
	set .@menu$, "";
	set .@cnt, 0;
	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		set .@it, @inventorylist_id[.@i];
		set .@ok, 0;
		if (.@slot == 1 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_WEAPON)
			set .@ok, 1;
		if (.@slot == 2 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_HAND_L))
			set .@ok, 1;
		if (.@slot == 3 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_HEAD_TOP))
			set .@ok, 1;
		if (.@slot == 4 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_ARMOR))
			set .@ok, 1;
		if (.@ok) {
			set .@cnt, .@cnt + 1;
			set .@id[.@cnt], .@it;
			set .@menu$, .@menu$ + getitemname(.@it);
			if (@inventorylist_amount[.@i] > 1)
				set .@menu$, .@menu$ + " x" + @inventorylist_amount[.@i];
			set .@menu$, .@menu$ + ":";
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have any suitable";
		mes "item in your inventory.";
		close;
	}
	set .@sel, select(.@menu$ + "Cancel");
	if (.@sel == .@cnt + 1)
		close;
	set .@item, .@id[.@sel];
	if (.@slot == 1 && #COMPANION_WEAPON[.@idx]) {
		getitem #COMPANION_WEAPON[.@idx], 1;
	} else if (.@slot == 2 && #COMPANION_SHIELD[.@idx]) {
		getitem #COMPANION_SHIELD[.@idx], 1;
	} else if (.@slot == 3 && #COMPANION_HELM[.@idx]) {
		getitem #COMPANION_HELM[.@idx], 1;
	} else if (.@slot == 4 && #COMPANION_ARMOR[.@idx]) {
		getitem #COMPANION_ARMOR[.@idx], 1;
	}
	delitem .@item, 1;
	if (.@slot == 1)
		set #COMPANION_WEAPON[.@idx], .@item;
	else if (.@slot == 2)
		set #COMPANION_SHIELD[.@idx], .@item;
	else if (.@slot == 3)
		set #COMPANION_HELM[.@idx], .@item;
	else
		set #COMPANION_ARMOR[.@idx], .@item;
	atcommand "@save";
	callsub OnEquipRefresh, .@idx;
	mes "[AI Companion]";
	mes #COMPANION_NAME$[.@idx] + " equipped";
	mes getitemname(.@item) + "!";
	close;

OnPCDieEvent:
	set .@priest, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i] != 1658 && #COMPANION_CLASS[.@i] != 1637)
			continue;
		if (!#COMPANION_GID[.@i] || !unitexists(#COMPANION_GID[.@i]))
			continue;
		set .@priest, 1;
		break;
	}
	if (!.@priest)
		end;
	if (#COMPANION_RES_TICK > gettimetick(2)) {
		dispbottom "Your priest companion needs more time to recover its divine power...";
		end;
	}
	set #COMPANION_RES_TICK, gettimetick(2) + 60;
	recovery 0, getcharid(0), 4;
	specialeffect2 EF_RESURRECTION;
	dispbottom "Your priest companion has resurrected you!";
	end;
