//===== rAthena Script ========================================
//= AI Companion
//= Hire up to 3 AI soldiers (slave mobs) that follow the player
//= and fight monsters alongside them.
//= Uses the native 'summon' command, no C++ changes needed.
//= Each companion is stored in an array slot (0..2).
//= Tune stat formulas inside OnScale / OnScaleFull.
//=============================================================
//= โครงสร้าง label หลัก ๆ:
//=   OnHireMenu      - เมนูจ้าง companion ใหม่ (เลือกสายอาชีพ 12 แบบ)
//=   OnRelease       - ปลด companion และคืนอุปกรณ์
//=   OnStatus        - ดูสถานะ (เลเวล / EXP / Alive)
//=   OnPickSlot      - เลือกว่าจะจัดการช่องไหน (subroutine)
//=   OnSummon        - เรียก companion ออกมาสู่แผนที่
//=   OnScale         - คำนวณสเตตตามเลเวล (รวมบัฟจากอุปกรณ์)
//=   OnScaleFull     - ตั้งค่าเต็มตัว (สไปรต์, HP, โหมด AI)
//=   OnCompanionTick - ตัวจับเวลา 5 วิ: ตามผู้เล่น / ฟื้นคืนชีพ / ให้บัฟ
//=   OnCompanionDead - เหตุการณ์เมื่อ companion ตาย
//=   OnCompanionBuff - ให้บัฟประจำสายอาชีพ (self + ผู้เล่น)
//=   OnNPCKillEvent  - เมื่อ companion ฆ่ามอนสเตอร์: เก็บ EXP
//=   OnVacuumLoot    - ดูดของดรอปมาให้ผู้เล่น
//=   OnNextExp       - คำนวณ EXP ถึงเลเวลถัดไป
//=   OnLevelUp       - เมื่อเลเวลอัป: ประกาศ + คำนวณสเตตใหม่
//=   OnEquipMenu     - เมนูใส่อุปกรณ์ (อาวุธ โล่ หัว เกราะ) + ปรับสเตต
//=   OnStatMenu      - ปรับสเตต (18 ค่า, 5 แต้ม/ครั้ง)
//=   OnEquipSlot     - เลือกไอเทมจากกระเป๋ามาใส่
//=   OnUnequipMenu   - ถอดอุปกรณ์คืนผู้เล่น
//=   OnViewEquip     - ดูอุปกรณ์ที่สวมอยู่
//=   OnEquipRefresh  - รีเฟรชสไปรต์/สเตตหลังใส่อุปกรณ์
//=   OnPCLoginEvent  - เมื่อผู้เล่น login: เรียก companion กลับมา
//=   OnPCDieEvent    - เมื่อผู้เล่นตาย: priest companion ปลุกให้
//=============================================================

//=== ตัวสคริปต์หลัก (ตอนเปิดเมนู) =============================
// เป็นเมนูแรกของ NPC AI Companion ที่ Prontera 156,193
// มี 4 ตัวเลือก: จ้าง / ปลด / ดูสถานะ / ใส่อุปกรณ์
// (ถ้าเปิดผ่าน @companion จากระยะไกลจะทำงานเหมือนกัน)
prontera,156,193,5	script	AI Companion	123,{
	mes "[AI Companion]";
	mes "Greetings, adventurer.";
	mes "I can summon up to 3 AI";
	mes "soldiers to be your";
	mes "companions. They will";
	mes "follow you and fight by";
	mes "your side, sharing the";
	mes "adventure with you!";
	next;
	switch(select("Hire a companion:Release companion:Check companions:Equip your companion")) {
	case 1:
		callsub OnHireMenu;
		close;
	case 2:
		callsub OnRelease;
		close;
	case 3:
		callsub OnStatus;
		close;
	case 4:
		callsub OnEquipMenu;
		close;
	}

//=== OnHireMenu: เมนูจ้าง companion ============================
// 1) นับจำนวน companion ที่มีอยู่ (0..2) ถ้าครบ 3 ให้บอกผู้เล่น
// 2) ให้เลือกสายอาชีพ 12 แบบ (แต่ละแบบมีสเตต/บัฟต่างกัน)
// 3) บันทึกข้อมูลลงตัวแปรถาวร #COMPANION_* แล้วเรียก OnSummon
// 4) ถ้าเรียกไม่สำเร็จ (summon ได้ 0) ให้ลบข้อมูลและบอกผู้เล่น
OnHireMenu:
	set .@slots, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1)
		if (#COMPANION_CLASS[.@i])
			set .@slots, .@slots + 1;
	if (.@slots >= 3) {
		mes "[AI Companion]";
		mes "Your companion party is";
		mes "full (3/3). Release one";
		mes "first if you want to";
		mes "hire a different one.";
		close;
	}
	mes "[AI Companion]";
	mes "You have " + .@slots + "/3 companions.";
	mes "Choose your next AI friend:";
	next;
	switch(select("Seyren - Lord Knight (tank):Eremes - Assassin (crit):Cenia - Priestess (magic):Howard - Creator (atk):Margaretha - High Priest (support):Cecil - Gunslinger (dex):Kathryne - High Wizard (nuke):Randel - Crusader (tank):Flamel - Paladin (holy tank):Celia - Wizard (fire):Chen - Monk (dmg):Gertie - Rogue (evade)")) {
	case 1:
		set .@cls, 1799;
		set .@nm$, "Seyren";
		break;
	case 2:
		set .@cls, 1800;
		set .@nm$, "Eremes";
		break;
	case 3:
		set .@cls, 1658;
		set .@nm$, "Cenia";
		break;
	case 4:
		set .@cls, 1636;
		set .@nm$, "Howard";
		break;
	case 5:
		set .@cls, 1637;
		set .@nm$, "Margaretha";
		break;
	case 6:
		set .@cls, 1638;
		set .@nm$, "Cecil";
		break;
	case 7:
		set .@cls, 1639;
		set .@nm$, "Kathryne";
		break;
	case 8:
		set .@cls, 3226;
		set .@nm$, "Randel";
		break;
	case 9:
		set .@cls, 3227;
		set .@nm$, "Flamel";
		break;
	case 10:
		set .@cls, 3228;
		set .@nm$, "Celia";
		break;
	case 11:
		set .@cls, 3229;
		set .@nm$, "Chen";
		break;
	case 12:
		set .@cls, 3230;
		set .@nm$, "Gertie";
		break;
	}
	set .@idx, -1;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (!#COMPANION_CLASS[.@i]) {
			set .@idx, .@i;
			break;
		}
	}
	set #COMPANION_CLASS[.@idx], .@cls;
	set #COMPANION_NAME$[.@idx], .@nm$;
	set #COMPANION_GID[.@idx], 0;
	set #COMPANION_DEAD_TICK[.@idx], 0;
	set #COMPANION_SKILLCD[.@idx], 0;
	if (!#COMPANION_LEVEL[.@cls])
		set #COMPANION_LEVEL[.@cls], 1;
	if (!#COMPANION_EXP[.@cls])
		set #COMPANION_EXP[.@cls], 0;
	callsub OnSummon, .@idx;
	if (!#COMPANION_GID[.@idx]) {
		set #COMPANION_CLASS[.@idx], 0;
		set #COMPANION_NAME$[.@idx], "";
		mes "[AI Companion]";
		mes "Failed to summon your friend...";
		mes "Please try again.";
		close;
	}
	mes "[AI Companion]";
	mes "Your AI companion " + #COMPANION_NAME$[.@idx];
	mes "has been summoned. It will";
	mes "follow you and fight by";
	mes "your side!";
	addtimer 5000, "AI Companion::OnCompanionTick";
	close;

//=== OnRelease: ปลด companion ==================================
// 1) เลือกช่องที่จะปลด (OnPickSlot) 2) ฆ่ามอนสเตอร์ (unitkill)
// 3) ล้างข้อมูลถาวรทั้งหมดในช่องนั้น 4) คืนอุปกรณ์ 4 ช่องเข้าสินค้า
OnRelease:
	callsub OnPickSlot;
	if (@pick_slot < 0)
		close;
	set .@idx, @pick_slot;
	if (#COMPANION_GID[.@idx])
		unitkill #COMPANION_GID[.@idx];
	set #COMPANION_CLASS[.@idx], 0;
	set #COMPANION_GID[.@idx], 0;
	set #COMPANION_DEAD_TICK[.@idx], 0;
	set #COMPANION_SKILLCD[.@idx], 0;
	set #COMPANION_NAME$[.@idx], "";
	if (#COMPANION_WEAPON[.@idx]) {
		getitem #COMPANION_WEAPON[.@idx], 1;
		set #COMPANION_WEAPON[.@idx], 0;
	}
	if (#COMPANION_SHIELD[.@idx]) {
		getitem #COMPANION_SHIELD[.@idx], 1;
		set #COMPANION_SHIELD[.@idx], 0;
	}
	if (#COMPANION_HELM[.@idx]) {
		getitem #COMPANION_HELM[.@idx], 1;
		set #COMPANION_HELM[.@idx], 0;
	}
	if (#COMPANION_ARMOR[.@idx]) {
		getitem #COMPANION_ARMOR[.@idx], 1;
		set #COMPANION_ARMOR[.@idx], 0;
	}
	atcommand "@save";
	mes "[AI Companion]";
	mes "Your companion has been released.";
	mes "Farewell, and stay safe!";
	close;

//=== OnStatus: ตรวจสอบสถานะ companion =========================
// วนลูป 3 ช่อง แสดง ชื่อ, เลเวล (สูงสุด 999), EXP ถึงเลเวลถัดไป,
// และสถานะ Alive (ยังอยู่ / กำลังฟื้นคืนชีพ)
OnStatus:
	set .@cnt, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set .@cnt, .@cnt + 1;
			mes "[AI Companion]";
			mes "Slot " + (.@i + 1) + " : " + #COMPANION_NAME$[.@i];
			mes "Level     : " + #COMPANION_LEVEL[#COMPANION_CLASS[.@i]] + " / 999";
			callsub OnNextExp, #COMPANION_LEVEL[#COMPANION_CLASS[.@i]];
			mes "EXP       : " + #COMPANION_EXP[#COMPANION_CLASS[.@i]] + " / " + @next_exp;
			if (#COMPANION_GID[.@i] && unitexists(#COMPANION_GID[.@i])) {
				set .@d[UMOB_MASTERAID], -1;
				getunitdata(#COMPANION_GID[.@i], .@d);
				if (.@d[UMOB_MASTERAID] == getcharid(3))
					mes "Status    : Alive";
				else
					mes "Status    : Recovering...";
			} else
				mes "Status    : Recovering...";
			next;
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have a companion.";
		close;
	}
	close;

//=== OnPickSlot: เลือกช่อง companion ==========================
// subroutine ย่อย: ให้ผู้เล่นเลือกว่าจะจัดการ companion ตัวไหน (จากช่อง 0..2)
// ถ้ามีตัวเดียวเลือกให้เลย ถ้าไม่มีเลยคืนค่า @pick_slot = -1
// ใช้ร่วมกับ OnRelease / OnEquipMenu
OnPickSlot:
	set .@menu$, "";
	set .@cnt, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set .@cnt, .@cnt + 1;
			set .@slot[.@cnt], .@i;
			set .@menu$, .@menu$ + #COMPANION_NAME$[.@i];
			if (.@cnt == 1)
				set .@menu$, .@menu$ + " (Lv " + #COMPANION_LEVEL[#COMPANION_CLASS[.@i]] + ")";
			set .@menu$, .@menu$ + ":";
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have a companion.";
		set @pick_slot, -1;
		return;
	}
	if (.@cnt == 1) {
		set @pick_slot, .@slot[1];
		return;
	}
	mes "[AI Companion]";
	mes "Choose a companion:";
	next;
	set .@sel, select(.@menu$ + "Cancel");
	if (.@sel == .@cnt + 1) {
		set @pick_slot, -1;
		return;
	}
	set @pick_slot, .@slot[.@sel];
	return;

//=== OnSummon: เรียก companion ออกมา ===========================
// ใช้คำสั่ง summon สร้างมอนสเตอร์ (ตัว companion) ตามคลาสที่บันทึกไว้
// แล้วจด GID ลงตัวแปรถาวร; ถ้า summon ไม่สำเร็จจะได้ 0 และจบงาน
OnSummon:
	set .@idx, getarg(0);
	set @gid, summon(#COMPANION_NAME$[.@idx], #COMPANION_CLASS[.@idx], 2147483647, "AI Companion::OnCompanionDead");
	set #COMPANION_GID[.@idx], @gid;
	if (!@gid)
		return;
	callsub OnScaleFull, .@idx;
	return;

//=== OnScale: คำนวณสเตตของ companion ตามเลเวล =================
// ตั้งค่า STR/AGI/VIT/INT/DEX/LUK, ATK, MATK, DEF, MDEF, HIT, FLEE, CRIT
// ตามสูตรเลเวล + บัฟประจำสายอาชีพ + ค่าที่ผู้เล่นปรับ (COMPANION_BS)
// แล้วบวกค่าอุปกรณ์ (อาวุธ/โล่/หัว/เกราะ) ที่สวมใส่
OnScale:
	set .@idx, getarg(0);
	if (#COMPANION_LEVEL[#COMPANION_CLASS[.@idx]] < 1)
		set #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]], 1;
	set @gid, #COMPANION_GID[.@idx];
	if (!@gid || !unitexists(@gid))
		return;
	set @lvl, #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]];
	setunitdata(@gid, UMOB_LEVEL, @lvl);
	setunitdata(@gid, UMOB_STR, 50 + @lvl * 2);
	setunitdata(@gid, UMOB_AGI, 40 + @lvl * 2);
	setunitdata(@gid, UMOB_VIT, 40 + @lvl * 2);
	setunitdata(@gid, UMOB_INT, 30 + @lvl * 2);
	setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
	setunitdata(@gid, UMOB_LUK, 25 + @lvl);
	setunitdata(@gid, UMOB_ATKMIN, @lvl * 20);
	setunitdata(@gid, UMOB_ATKMAX, @lvl * 28);
	setunitdata(@gid, UMOB_DEF, @lvl / 2 + 10);
	setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 10);
	setunitdata(@gid, UMOB_HIT, 100 + @lvl);
	setunitdata(@gid, UMOB_FLEE, 80 + @lvl);
	setunitdata(@gid, UMOB_CRIT, 20);
	if (#COMPANION_CLASS[.@idx] == 1799) {
		setunitdata(@gid, UMOB_VIT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_STR, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 1800) {
		setunitdata(@gid, UMOB_AGI, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_LUK, 40 + @lvl * 2);
		setunitdata(@gid, UMOB_CRIT, 40);
		setunitdata(@gid, UMOB_FLEE, 100 + @lvl);
	} else if (#COMPANION_CLASS[.@idx] == 1658 || #COMPANION_CLASS[.@idx] == 3228) {
		setunitdata(@gid, UMOB_INT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 18);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 26);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 16);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 22);
	} else if (#COMPANION_CLASS[.@idx] == 1636) {
		setunitdata(@gid, UMOB_STR, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 24);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 1637) {
		setunitdata(@gid, UMOB_INT, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_VIT, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 12);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 18);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 1638) {
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_AGI, 50 + @lvl * 2);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 22);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 30);
	} else if (#COMPANION_CLASS[.@idx] == 1639) {
		setunitdata(@gid, UMOB_INT, 70 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 22);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 3226) {
		setunitdata(@gid, UMOB_VIT, 55 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
	} else if (#COMPANION_CLASS[.@idx] == 3227) {
		setunitdata(@gid, UMOB_VIT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_STR, 55 + @lvl * 2);
		setunitdata(@gid, UMOB_DEF, @lvl / 2 + 30);
		setunitdata(@gid, UMOB_MDEF, @lvl / 2 + 25);
	} else if (#COMPANION_CLASS[.@idx] == 3229) {
		setunitdata(@gid, UMOB_STR, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_AGI, 50 + @lvl * 2);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 24);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 32);
	} else if (#COMPANION_CLASS[.@idx] == 3230) {
		setunitdata(@gid, UMOB_AGI, 65 + @lvl * 3);
		setunitdata(@gid, UMOB_DEX, 120 + @lvl * 3);
		setunitdata(@gid, UMOB_FLEE, 110 + @lvl);
		setunitdata(@gid, UMOB_CRIT, 30);
	} else {
		setunitdata(@gid, UMOB_INT, 60 + @lvl * 3);
		setunitdata(@gid, UMOB_MATKMIN, @lvl * 18);
		setunitdata(@gid, UMOB_MATKMAX, @lvl * 26);
		setunitdata(@gid, UMOB_ATKMIN, @lvl * 16);
		setunitdata(@gid, UMOB_ATKMAX, @lvl * 22);
	}
	if (#COMPANION_BS[.@idx * 20 + 1])
		setunitdata(@gid, UMOB_STR, #COMPANION_BS[.@idx * 20 + 1] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 1]));
	if (#COMPANION_BS[.@idx * 20 + 2])
		setunitdata(@gid, UMOB_AGI, #COMPANION_BS[.@idx * 20 + 2] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 2]));
	if (#COMPANION_BS[.@idx * 20 + 3])
		setunitdata(@gid, UMOB_VIT, #COMPANION_BS[.@idx * 20 + 3] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 3]));
	if (#COMPANION_BS[.@idx * 20 + 4])
		setunitdata(@gid, UMOB_INT, #COMPANION_BS[.@idx * 20 + 4] + 2 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 4]));
	if (#COMPANION_BS[.@idx * 20 + 5])
		setunitdata(@gid, UMOB_DEX, #COMPANION_BS[.@idx * 20 + 5] + 3 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 5]));
	if (#COMPANION_BS[.@idx * 20 + 6])
		setunitdata(@gid, UMOB_LUK, #COMPANION_BS[.@idx * 20 + 6] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 6]));
	if (#COMPANION_BS[.@idx * 20 + 7]) {
		setunitdata(@gid, UMOB_ATKMIN, #COMPANION_BS[.@idx * 20 + 7] + 24 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 7]));
		setunitdata(@gid, UMOB_ATKMAX, (#COMPANION_BS[.@idx * 20 + 7] + 24 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 7])) * 4 / 3);
	}
	if (#COMPANION_BS[.@idx * 20 + 8]) {
		setunitdata(@gid, UMOB_MATKMIN, #COMPANION_BS[.@idx * 20 + 8] + 20 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 8]));
		setunitdata(@gid, UMOB_MATKMAX, (#COMPANION_BS[.@idx * 20 + 8] + 20 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 8])) * 3 / 2);
	}
	if (#COMPANION_BS[.@idx * 20 + 10])
		setunitdata(@gid, UMOB_DEF, #COMPANION_BS[.@idx * 20 + 10] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 10]));
	if (#COMPANION_BS[.@idx * 20 + 11])
		setunitdata(@gid, UMOB_MDEF, #COMPANION_BS[.@idx * 20 + 11] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 11]));
	if (#COMPANION_BS[.@idx * 20 + 16])
		setunitdata(@gid, UMOB_PDODGE, #COMPANION_BS[.@idx * 20 + 16] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 16]));
	if (#COMPANION_BS[.@idx * 20 + 17])
		setunitdata(@gid, UMOB_RES, #COMPANION_BS[.@idx * 20 + 17] + 1 * (@lvl - #COMPANION_BSLV[.@idx * 20 + 17]));
	if (#COMPANION_BS[.@idx * 20 + 18])

//=== OnCompanionTick: ตัวจับเวลา 5 วินาที =======================
// ทำงานทุก 5 วิ (เรียกซ้ำด้วย addtimer) สำหรับ companion ที่ยังมี:
//  - ถ้า companion ยังอยู่และเป็นของเรา: คำนวณสเตตใหม่, ให้บัฟ, 
//    และวาร์ปตามผู้เล่นถ้าอยู่ไกล (>15 ช่อง) หรือคนละแผนที่
//  - ถ้า companion ตาย: ตั้งเวลา DEAD_TICK และฟื้นคืนชีพอัตโนมัติ
//    หลัง 10 วิ (รอ cooldown) ด้วย OnSummon
OnCompanionTick:
	set @has_any, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set @has_any, 1;
			if (#COMPANION_GID[.@i] && unitexists(#COMPANION_GID[.@i])) {
				set .@d[UMOB_MASTERAID], -1;
				set .@d[UMOB_MAPID], -1;
				set .@d[UMOB_X], -1;
				set .@d[UMOB_Y], -1;
				getunitdata(#COMPANION_GID[.@i], .@d);
				if (.@d[UMOB_MASTERAID] == getcharid(3) && .@d[UMOB_MAPID] >= 0) {
					callsub OnScale, .@i;
					callsub OnCompanionBuff, #COMPANION_CLASS[.@i], #COMPANION_GID[.@i];
					callsub OnCompanionSkill, .@i, #COMPANION_CLASS[.@i], #COMPANION_GID[.@i];
					getmapxy(@comp_map$, @comp_x, @comp_y, BL_PC);
					if (@comp_x != -1 && @comp_y != -1) {
						set @df, distance(@comp_x, @comp_y, .@d[UMOB_X], .@d[UMOB_Y]);
						if (.@d[UMOB_MAPID] != mapname2id(strcharinfo(3)) || @df > 15)
							unitwarp(#COMPANION_GID[.@i], @comp_map$, @comp_x, @comp_y);
					}
				} else {
					set #COMPANION_GID[.@i], 0;
					if (#COMPANION_DEAD_TICK[.@i] == 0) {
						set #COMPANION_DEAD_TICK[.@i], gettimetick(2);
						dispbottom #COMPANION_NAME$[.@i] + " has fallen...";
					}
					if (gettimetick(2) - #COMPANION_DEAD_TICK[.@i] < 10) {
						// wait for revive cooldown
					} else {
						callsub OnSummon, .@i;
						if (#COMPANION_GID[.@i]) {
							set #COMPANION_DEAD_TICK[.@i], 0;
							dispbottom #COMPANION_NAME$[.@i] + " has returned!";
						}
					}
				}
			} else {
				if (#COMPANION_DEAD_TICK[.@i] == 0) {
					set #COMPANION_DEAD_TICK[.@i], gettimetick(2);
					dispbottom #COMPANION_NAME$[.@i] + " has fallen...";
				}
				if (gettimetick(2) - #COMPANION_DEAD_TICK[.@i] < 10) {
					// wait for revive cooldown
				} else {
					callsub OnSummon, .@i;
					if (#COMPANION_GID[.@i]) {
						set #COMPANION_DEAD_TICK[.@i], 0;
						dispbottom #COMPANION_NAME$[.@i] + " has returned!";
					}
				}
			}
		}
	}
	if (@has_any)
		addtimer 5000, "AI Companion::OnCompanionTick";
	end;

//=== OnCompanionDead: เหตุการณ์ตอน companion ตาย ===============

OnCompanionDead:
	end;

//=== OnCompanionSkill: สกิลโจมตีประจำสายอาชีพ ===================
// รับพารามิเตอร์: .@i (ช่อง), .@cls (คลาส), .@gid (GID companion)
// 1) เช็ค cooldown 8 วิ ต่อช่อง (#COMPANION_SKILLCD)
// 2) เลือกสกิลตามสายอาชีพจากตารางด้านล่าง (มีทั้งสกิลเดี่ยว/AoE)
//    - สกิลแบบ self-AoE (Magnum/Meteor Assault/Grand Cross)
//      ให้ร่ายใส่ตัวเองเพื่อโจมตีรอบตัว
//    - สกิลแบบยิงเป้า/Ground จะยิงใส่ศัตรู (ตกลงที่ตำแหน่งศัตรู)
// 3) หาเป้าหมาย: ใช้เป้าหมายที่ companion โจมตีอยู่ (UMOB_TARGETID)
//    ถ้าไม่มี ให้ scan mob ระยะ 9 ช่องรอบผู้เล่น
// 4) ร่ายด้วย unitskilluseid / unitskillusepos (ยกเลิกไม่ได้)
//    - สกิลเป้าแบบ Ground (Magnus/Storm Gust/Meteor) ต้องใช้ unitskillusepos
//      ยิงที่ตำแหน่งเป้า (หน่วยเซิร์ฟเวอร์ลง timer เป็น skill_castend_pos
//      มิฉะนั้นตอน cast ถูกขัดจะ error "delete_timer function mismatch")
// 5) ค่าชดเชย cast time (.@cc): เวลาร่ายจริง = .@cc x 1000ms + cast time
//    ดั้งเดิมของสกิล (ยิงผ่าน unitskill* เสมอ) จึงใส่ค่าติดลบสำหรับสกิล
//    ที่มี cast time มาก (Magnus 10s / Storm Gust 6.3s / Grand Cross 1s)
//    เพื่อให้ร่ายไว ไม่ยืนค้าง
OnCompanionSkill:
	set .@i, getarg(0);
	set .@cls, getarg(1);
	set .@gid, getarg(2);
	if (!.@gid || !unitexists(.@gid))
		return;
	if (gettimetick(2) - #COMPANION_SKILLCD[.@i] < 8)
		return;
	set .@sk$, "";
	set .@slv, 0;
	set .@self, 0;
	set .@ground, 0;
	set .@cc, 0;
	switch(.@cls) {
	case 1799:
		set .@sk$, "KN_BOWLINGBASH";
		set .@slv, 10;
		break;
	case 1800:
		set .@sk$, "ASC_METEORASSAULT";
		set .@slv, 10;
		set .@self, 1;
		break;
	case 1658:
		set .@sk$, "AL_HOLYLIGHT";
		set .@slv, 10;
		break;
	case 1636:
		set .@sk$, "SM_MAGNUM";
		set .@slv, 10;
		set .@self, 1;
		break;
	case 1637:
		set .@sk$, "PR_MAGNUS";
		set .@slv, 10;
		set .@ground, 1;
		set .@cc, -9;
		break;
	case 1638:
		set .@sk$, "GS_PIERCINGSHOT";
		set .@slv, 10;
		break;
	case 1639:
		set .@sk$, "WZ_STORMGUST";
		set .@slv, 10;
		set .@ground, 1;
		set .@cc, -6;
		break;
	case 3226:
		set .@sk$, "CR_SHIELDBOOMERANG";
		set .@slv, 10;
		break;
	case 3227:
		set .@sk$, "CR_GRANDCROSS";
		set .@slv, 10;
		set .@self, 1;
		set .@cc, -1;
		break;
	case 3228:
		set .@sk$, "WZ_METEOR";
		set .@slv, 10;
		set .@ground, 1;
		break;
	case 3229:
		set .@sk$, "MO_FINGEROFFENSIVE";
		set .@slv, 10;
		break;
	case 3230:
		set .@sk$, "AS_GRIMTOOTH";
		set .@slv, 10;
		break;
	}
	if (!.@slv)
		return;
	getmapxy(@comp_map$, @comp_x, @comp_y, BL_PC);
	if (@comp_x == -1)
		return;
	if (.@self) {
		set #COMPANION_SKILLCD[.@i], gettimetick(2);
		unitskilluseid .@gid, .@sk$, .@slv, .@gid, .@cc, 1, 0, 1;
		return;
	}
	set .@d[UMOB_X], -1;
	set .@d[UMOB_Y], -1;
	set .@d[UMOB_TARGETID], 0;
	getunitdata(.@gid, .@d);
	set .@tg, .@d[UMOB_TARGETID];
	set .@has_tg, 0;
	if (.@tg && unitexists(.@tg)) {
		set .@d2[UMOB_MAPID], -1;
		getunitdata(.@tg, .@d2);
		if (.@d2[UMOB_MAPID] == mapname2id(@comp_map$))
			set .@has_tg, 1;
	}
	if (!.@has_tg) {
		set .@n, getmapunits(BL_MOB, @comp_map$, .@mob[0]);
		for (set .@k, 0; .@k < .@n; set .@k, .@k + 1) {
			if (.@mob[.@k] == .@gid)
				continue;
			set .@d2[UMOB_X], -1;
			set .@d2[UMOB_Y], -1;
			getunitdata(.@mob[.@k], .@d2);
			if (.@d2[UMOB_X] == -1)
				continue;
			set .@dd, distance(.@d2[UMOB_X], .@d2[UMOB_Y], @comp_x, @comp_y);
			if (.@dd <= 9) {
				set .@tg, .@mob[.@k];
				break;
			}
		}
	}
	if (!.@tg)
		return;
	set .@d2[UMOB_X], -1;
	set .@d2[UMOB_Y], -1;
	getunitdata(.@tg, .@d2);
	set #COMPANION_SKILLCD[.@i], gettimetick(2);
	if (.@ground) {
		unitskillusepos .@gid, .@sk$, .@slv, .@d2[UMOB_X], .@d2[UMOB_Y], .@cc, 1, 0, 1;
		return;
	}
	unitskilluseid .@gid, .@sk$, .@slv, .@tg, .@cc, 1, 0, 1;
	return;

//=== OnNPCKillEvent: เมื่อ companion ฆ่ามอนสเตอร์ ================
// event ติดตั้งที่ NPC นี้ (เรียกเมื่อ kill); ตรวจหา companion ที่
// เป็นตัวสังหาร (killerrid == #COMPANION_GID) แล้ว:
//  - รับ EXP = เลเวลมอนสเตอร์ x 15 + 50 เก็บลง #COMPANION_EXP
//  - วนอัปเลเวลถ้า EXP เต็ม (สูงสุด 999) และเรียก OnLevelUp
//  - สุดท้ายดูดของที่ดรอปมาให้ผู้เล่น (OnVacuumLoot)
OnNPCKillEvent:
	set @comp_killed, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (!#COMPANION_CLASS[.@i])
			continue;
		if (#COMPANION_LEVEL[#COMPANION_CLASS[.@i]] >= 999)
			continue;
		if (!#COMPANION_GID[.@i])
			continue;
		if (#COMPANION_GID[.@i] != killerrid)
			continue;
		if (!unitexists(#COMPANION_GID[.@i]))
			continue;
		set .@d[UMOB_MASTERAID], -1;
		set .@d[UMOB_X], -1;
		set .@d[UMOB_Y], -1;
		getunitdata(#COMPANION_GID[.@i], .@d);
		if (.@d[UMOB_MASTERAID] != getcharid(3))
			continue;
		set @comp_killed, 1;
		set @kill_x, .@d[UMOB_X];
		set @kill_y, .@d[UMOB_Y];
		set @moblv, getmonsterinfo(killedrid, MOB_LV);
		if (@moblv < 1)
			set @moblv, 1;
		set .@class, #COMPANION_CLASS[.@i];
		set .@lvl, #COMPANION_LEVEL[.@class];
		set .@exp, #COMPANION_EXP[.@class] + @moblv * 15 + 50;
		callsub OnNextExp, .@lvl;
		while (.@lvl < 999 && .@exp >= @next_exp) {
			set .@exp, .@exp - @next_exp;
			set .@lvl, .@lvl + 1;
			if (.@lvl <= BaseLevel)
				set @next_exp, getbaseexp_ratio(100, .@lvl);
			else
				set @next_exp, .@lvl * .@lvl * 5 + .@lvl * 100;
		}
		set #COMPANION_EXP[.@class], .@exp;
		set #COMPANION_LEVEL[.@class], .@lvl;
		callsub OnLevelUp, .@i;
		break;
	}
	if (@comp_killed)
		callsub OnVacuumLoot, @kill_x, @kill_y;
	end;

//=== OnVacuumLoot: ดูดของที่ดรอปมาให้ผู้เล่น =====================
// scan หาไอเทม (BL_ITEM) บนแผนที่รอบจุดที่ companion ฆ่ามอนสเตอร์
// (ระยะ <= 5 ช่อง) แล้ววาร์ปทั้งหมดมาไว้ที่ตัวผู้เล่น
OnVacuumLoot:
	set .@kx, getarg(0);
	set .@ky, getarg(1);
	getmapxy(@v_map$, @v_px, @v_py, BL_PC);
	if (@v_px == -1)
		return;
	set .@num, getmapunits(BL_ITEM, @v_map$, .@ig[0]);
	for (set .@k, 0; .@k < .@num; set .@k, .@k + 1) {
		set .@d[UMOB_X], -1;
		set .@d[UMOB_Y], -1;
		getunitdata(.@ig[.@k], .@d);
		if (.@d[UMOB_X] == -1 || .@d[UMOB_Y] == -1)
			continue;
		set .@dd, distance(.@d[UMOB_X], .@d[UMOB_Y], .@kx, .@ky);
		if (.@dd <= 5)
			unitwarp(.@ig[.@k], @v_map$, @v_px, @v_py);
	}
	return;

//=== OnNextExp: คำนวณ EXP ที่ต้องใช้ถึงเลเวลถัดไป ================
// subroutine ย่อย คืนค่า @next_exp จากเลเวลปัจจุบัน:
//  - เลเวลต่ำกว่า BaseLevel: ใช้สูตรของเซิร์ฟเวอร์ (getbaseexp_ratio)
//  - เลเวลสูงกว่านั้น: ใช้สูตรเอง lvl^2*5 + lvl*100 (เร่งจบ 999)
OnNextExp:
	set .@lvl, getarg(0);
	if (.@lvl <= BaseLevel)
		set @next_exp, getbaseexp_ratio(100, .@lvl);
	else
		set @next_exp, .@lvl * .@lvl * 5 + .@lvl * 100;
	return;

//=== OnLevelUp: เมื่อ companion เลเวลอัป =========================
// แจ้งผู้เล่นด้วย dispbottom ว่า companion อัปเลเวล แล้วคำนวณ
// สเตต/HP ใหม่ทั้งชุด (OnScaleFull) ให้ค่าต่างๆ ตามเลเวลใหม่ทันที
OnLevelUp:
	set .@idx, getarg(0);
	dispbottom "Your AI companion " + #COMPANION_NAME$[.@idx] + " reached level " + #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]] + "!";
	callsub OnScaleFull, .@idx;
	return;

//=== OnEquipMenu: เมนูจัดการอุปกรณ์ของ companion =================
// เลือก companion ก่อน (OnPickSlot) แล้วให้เลือก:
//  1-4) ใส่ Weapon/Shield/Headgear/Armor (OnEquipSlot)
//  5)   ถอดอุปกรณ์คืนผู้เล่น (OnUnequipMenu)
//  6)   ดูอุปกรณ์ที่สวมอยู่ (OnViewEquip)
//  7)   ปรับสเตต (OnStatMenu)
OnEquipMenu:
	callsub OnPickSlot;
	if (@pick_slot < 0)
		close;
	set .@idx, @pick_slot;
	mes "[AI Companion]";
	mes "What would you like to equip";
	mes "on " + #COMPANION_NAME$[.@idx] + "?";
	next;
	switch(select("Weapon:Shield:Headgear:Armor:Unequip an item:View current equipment:Adjust Stats")) {
	case 1:
		callsub OnEquipSlot, .@idx, 1;
		close;
	case 2:
		callsub OnEquipSlot, .@idx, 2;
		close;
	case 3:
		callsub OnEquipSlot, .@idx, 3;
		close;
	case 4:
		callsub OnEquipSlot, .@idx, 4;
		close;
	case 5:
		callsub OnUnequipMenu, .@idx;
		close;
	case 6:
		callsub OnViewEquip, .@idx;
		close;
	case 7:
		callsub OnStatMenu, .@idx;
		close;
	}

//=== OnStatMenu: ปรับสเตตของ companion ==========================
// สร้างเมนู 18 สเตต (STR..MRES) แสดงค่าปัจจุบัน (รวมสูตรตามเลเวล)
// ให้ผู้เล่นเลือกเพิ่มสเตต 5 แต้ม/ครั้ง (ดู @bs_name$[1..18])
// หรือ Reset all stats; เก็บค่าลง #COMPANION_BS แล้วเรียก OnScale
// เพื่ออัปเดตสเตตจริงบนตัว companion ทันที
OnStatMenu:
	set .@idx, getarg(0);
	set @bs_name$[1], "STR";
	set @bs_name$[2], "AGI";
	set @bs_name$[3], "VIT";
	set @bs_name$[4], "INT";
	set @bs_name$[5], "DEX";
	set @bs_name$[6], "LUK";
	set @bs_name$[7], "ATK";
	set @bs_name$[8], "MATK";
	set @bs_name$[9], "MaxHP";
	set @bs_name$[10], "DEF";
	set @bs_name$[11], "MDEF";
	set @bs_name$[12], "HIT";
	set @bs_name$[13], "FLEE";
	set @bs_name$[14], "Crit";
	set @bs_name$[15], "ASPD";
	set @bs_name$[16], "PDodge";
	set @bs_name$[17], "RES";
	set @bs_name$[18], "MRES";
	set .@lvl, #COMPANION_LEVEL[#COMPANION_CLASS[.@idx]];
	set .@m$, "";
	for (set .@s, 1; .@s <= 18; set .@s, .@s + 1) {
		if (#COMPANION_BS[.@idx * 20 + .@s]) {
			set .@cur, 0;
			if (.@s == 1) set .@cur, #COMPANION_BS[.@idx * 20 + 1] + 2 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 1]);
			if (.@s == 2) set .@cur, #COMPANION_BS[.@idx * 20 + 2] + 2 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 2]);
			if (.@s == 3) set .@cur, #COMPANION_BS[.@idx * 20 + 3] + 2 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 3]);
			if (.@s == 4) set .@cur, #COMPANION_BS[.@idx * 20 + 4] + 2 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 4]);
			if (.@s == 5) set .@cur, #COMPANION_BS[.@idx * 20 + 5] + 3 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 5]);
			if (.@s == 6) set .@cur, #COMPANION_BS[.@idx * 20 + 6] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 6]);
			if (.@s == 7) set .@cur, #COMPANION_BS[.@idx * 20 + 7] + 24 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 7]);
			if (.@s == 8) set .@cur, #COMPANION_BS[.@idx * 20 + 8] + 20 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 8]);
			if (.@s == 9) set .@cur, #COMPANION_BS[.@idx * 20 + 9] + 260 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 9]);
			if (.@s == 10) set .@cur, #COMPANION_BS[.@idx * 20 + 10] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 10]);
			if (.@s == 11) set .@cur, #COMPANION_BS[.@idx * 20 + 11] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 11]);
			if (.@s == 12) set .@cur, #COMPANION_BS[.@idx * 20 + 12] + 3 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 12]);
			if (.@s == 13) set .@cur, #COMPANION_BS[.@idx * 20 + 13] + 2 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 13]);
			if (.@s == 14) set .@cur, #COMPANION_BS[.@idx * 20 + 14] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 14]);
			if (.@s == 15) set .@cur, #COMPANION_BS[.@idx * 20 + 15];
			if (.@s == 16) set .@cur, #COMPANION_BS[.@idx * 20 + 16] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 16]);
			if (.@s == 17) set .@cur, #COMPANION_BS[.@idx * 20 + 17] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 17]);
			if (.@s == 18) set .@cur, #COMPANION_BS[.@idx * 20 + 18] + 1 * (.@lvl - #COMPANION_BSLV[.@idx * 20 + 18]);
			set .@m$, .@m$ + @bs_name$[.@s] + " [" + .@cur + "]:";
		} else
			set .@m$, .@m$ + @bs_name$[.@s] + ":";
	}
	set .@m$, .@m$ + "Reset all stats:Cancel";
	mes "[AI Companion]";
	mes "Adjust stats of " + #COMPANION_NAME$[.@idx];
	mes "Current level: " + .@lvl;
	mes "Value = stat at this level,";
	mes "grows every level up.";
	mes "(0 = use default)";
	next;
	set .@sel, select(.@m$);
	if (.@sel == 20)
		close;
	if (.@sel == 19) {
		for (set .@s, 1; .@s <= 18; set .@s, .@s + 1) {
			set #COMPANION_BS[.@idx * 20 + .@s], 0;
			set #COMPANION_BSLV[.@idx * 20 + .@s], 0;
		}
		callsub OnScaleFull, .@idx;
		mes "[AI Companion]";
		mes "All custom stats reset.";
		close;
	}
	mes "[AI Companion]";
	mes "Enter value for " + @bs_name$[.@sel] + " at level " + .@lvl + ":";
	mes "(0 = use default)";
	next;
	input .@v;
	if (.@v < 0)
		set .@v, 0;
	if (.@v > 99999)
		set .@v, 99999;
	if (.@v > 0) {
		set #COMPANION_BS[.@idx * 20 + .@sel], .@v;
		set #COMPANION_BSLV[.@idx * 20 + .@sel], .@lvl;
	}
	callsub OnScaleFull, .@idx;
	mes "[AI Companion]";
	if (.@v > 0)
		mes @bs_name$[.@sel] + " set to " + .@v + " at lv " + .@lvl + ".";
	else
		mes @bs_name$[.@sel] + " reset to default.";
	close;

//=== OnEquipSlot: ใส่ไอเทมให้ companion ==========================
// รับพารามิเตอร์: .@idx (ช่อง) + .@slot (1=อาวุธ 2=โล่ 3=หัว 4=เกราะ)
// กรองไอเทมในกระเป๋าผู้เล่นให้ถูกชนิด/ตำแหน่ง (IT_WEAPON / EQP_*)
// แสดงเมนูให้เลือก; ถ้าแทนของเก่าคืนของเก่าให้ผู้เล่นก่อน (getitem)
// แล้ว delitem ของใหม่ เก็บ ID ลงตัวแปรถาวร และรีเฟรชสไปรต์
OnEquipSlot:
	set .@idx, getarg(0);
	set .@slot, getarg(1);
	getinventorylist();
	set .@menu$, "";
	set .@cnt, 0;
	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		set .@it, @inventorylist_id[.@i];
		set .@ok, 0;
		if (.@slot == 1 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_WEAPON)
			set .@ok, 1;
		if (.@slot == 2 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_HAND_L))
			set .@ok, 1;
		if (.@slot == 3 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_HEAD_TOP))
			set .@ok, 1;
		if (.@slot == 4 && getiteminfo(.@it, ITEMINFO_TYPE) == IT_ARMOR && (getiteminfo(.@it, ITEMINFO_LOCATIONS) & EQP_ARMOR))
			set .@ok, 1;
		if (.@ok) {
			set .@cnt, .@cnt + 1;
			set .@id[.@cnt], .@it;
			set .@menu$, .@menu$ + getitemname(.@it);
			if (@inventorylist_amount[.@i] > 1)
				set .@menu$, .@menu$ + " x" + @inventorylist_amount[.@i];
			set .@menu$, .@menu$ + ":";
		}
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "You don't have any suitable";
		mes "item in your inventory.";
		close;
	}
	set .@sel, select(.@menu$ + "Cancel");
	if (.@sel == .@cnt + 1)
		close;
	set .@item, .@id[.@sel];
	if (.@slot == 1 && #COMPANION_WEAPON[.@idx]) {
		getitem #COMPANION_WEAPON[.@idx], 1;
	} else if (.@slot == 2 && #COMPANION_SHIELD[.@idx]) {
		getitem #COMPANION_SHIELD[.@idx], 1;
	} else if (.@slot == 3 && #COMPANION_HELM[.@idx]) {
		getitem #COMPANION_HELM[.@idx], 1;
	} else if (.@slot == 4 && #COMPANION_ARMOR[.@idx]) {
		getitem #COMPANION_ARMOR[.@idx], 1;
	}
	delitem .@item, 1;
	if (.@slot == 1)
		set #COMPANION_WEAPON[.@idx], .@item;
	else if (.@slot == 2)
		set #COMPANION_SHIELD[.@idx], .@item;
	else if (.@slot == 3)
		set #COMPANION_HELM[.@idx], .@item;
	else
		set #COMPANION_ARMOR[.@idx], .@item;
	atcommand "@save";
	callsub OnEquipRefresh, .@idx;
	mes "[AI Companion]";
	mes #COMPANION_NAME$[.@idx] + " equipped";
	mes getitemname(.@item) + "!";
	close;

//=== OnUnequipMenu: ถอดอุปกรณ์คืนผู้เล่น ==========================
// สร้างเมนูจากของที่ companion สวมอยู่ (อาวุธ/โล่/หัว/เกราะ)
// ให้ผู้เล่นเลือกอันที่อยากถอด; คืนไอเทมเข้าสินค้า (getitem),
// ล้างค่า slot นั้น แล้วรีเฟรชสไปรต์ให้อัปเดตทันที
OnUnequipMenu:
	set .@idx, getarg(0);
	set .@menu$, "";
	set .@cnt, 0;
	if (#COMPANION_WEAPON[.@idx]) {
		set .@cnt, .@cnt + 1;
		set .@slot[.@cnt], 1;
		set .@menu$, .@menu$ + getitemname(#COMPANION_WEAPON[.@idx]) + " (Weapon):";
	}
	if (#COMPANION_SHIELD[.@idx]) {
		set .@cnt, .@cnt + 1;
		set .@slot[.@cnt], 2;
		set .@menu$, .@menu$ + getitemname(#COMPANION_SHIELD[.@idx]) + " (Shield):";
	}
	if (#COMPANION_HELM[.@idx]) {
		set .@cnt, .@cnt + 1;
		set .@slot[.@cnt], 3;
		set .@menu$, .@menu$ + getitemname(#COMPANION_HELM[.@idx]) + " (Headgear):";
	}
	if (#COMPANION_ARMOR[.@idx]) {
		set .@cnt, .@cnt + 1;
		set .@slot[.@cnt], 4;
		set .@menu$, .@menu$ + getitemname(#COMPANION_ARMOR[.@idx]) + " (Armor):";
	}
	if (!.@cnt) {
		mes "[AI Companion]";
		mes "Your companion has no";
		mes "equipment right now.";
		close;
	}
	set .@sel, select(.@menu$ + "Cancel");
	if (.@sel == .@cnt + 1)
		close;
	if (.@slot[.@sel] == 1) {
		getitem #COMPANION_WEAPON[.@idx], 1;
		set #COMPANION_WEAPON[.@idx], 0;
	} else if (.@slot[.@sel] == 2) {
		getitem #COMPANION_SHIELD[.@idx], 1;
		set #COMPANION_SHIELD[.@idx], 0;
	} else if (.@slot[.@sel] == 3) {
		getitem #COMPANION_HELM[.@idx], 1;
		set #COMPANION_HELM[.@idx], 0;
	} else {
		getitem #COMPANION_ARMOR[.@idx], 1;
		set #COMPANION_ARMOR[.@idx], 0;
	}
	atcommand "@save";
	callsub OnEquipRefresh, .@idx;
	mes "[AI Companion]";
	mes "Item returned to your";
	mes "inventory.";
	close;

//=== OnViewEquip: ดูอุปกรณ์ที่สวมอยู่ =============================
// แสดงรายการอาวุธ/โล่/หัว/เกราะที่ companion ใส่อยู่
// (ถ้าช่องไหนว่างแสดง "-")
OnViewEquip:
	set .@idx, getarg(0);
	mes "[AI Companion]";
	mes "Equipment of " + #COMPANION_NAME$[.@idx] + ":";
	mes "Weapon   : " + (#COMPANION_WEAPON[.@idx] ? getitemname(#COMPANION_WEAPON[.@idx]) : "-");
	mes "Shield   : " + (#COMPANION_SHIELD[.@idx] ? getitemname(#COMPANION_SHIELD[.@idx]) : "-");
	mes "Headgear : " + (#COMPANION_HELM[.@idx] ? getitemname(#COMPANION_HELM[.@idx]) : "-");
	mes "Armor    : " + (#COMPANION_ARMOR[.@idx] ? getitemname(#COMPANION_ARMOR[.@idx]) : "-");
	close;

//=== OnEquipRefresh: รีเฟรชสไปรต์/สเตตหลังใส่อุปกรณ์ ============
// ตรวจว่าการ์ดยังอยู่และเป็นของเราก่อน; ถ้าอยู่ เรียก OnScaleFull
// เพื่อให้ภาพอุปกรณ์ + สเตตใหม่ (จากไอเทมใหม่) ลงบนตัวทันที
OnEquipRefresh:
	set .@idx, getarg(0);
	if (!#COMPANION_GID[.@idx])
		return;
	if (!unitexists(#COMPANION_GID[.@idx]))
		return;
	set .@d[UMOB_MASTERAID], -1;
	getunitdata(#COMPANION_GID[.@idx], .@d);
	if (.@d[UMOB_MASTERAID] != getcharid(3))
		return;
	callsub OnScaleFull, .@idx;
	return;

//=== OnPCLoginEvent: เมื่อผู้เล่น login ============================
// เรียก companion ทั้ง 3 ช่องกลับมาที่แผนที่ปัจจุบัน (ล้าง GID เก่า
// แล้ว OnSummon ใหม่) และเริ่มตัวจับเวลา 5 วิ (OnCompanionTick)
OnPCLoginEvent:
	set @has_any, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i]) {
			set @has_any, 1;
			set #COMPANION_GID[.@i], 0;
			set #COMPANION_DEAD_TICK[.@i], 0;
			callsub OnSummon, .@i;
		}
	}
	if (@has_any)
		addtimer 5000, "AI Companion::OnCompanionTick";
	end;

//=== OnPCDieEvent: เมื่อผู้เล่นตาย ================================
// ถ้ามี priest companion (คลาส 1658/1637) อยู่ จะปลุกผู้เล่นด้วย
// recovery + เอฟเฟกต์ Resurrection; มี cooldown 60 วิ (RES_TICK)
// เพื่อกันใช้ซ้ำถี่เกินไป
OnPCDieEvent:
	set .@priest, 0;
	for (set .@i, 0; .@i < 3; set .@i, .@i + 1) {
		if (#COMPANION_CLASS[.@i] != 1658 && #COMPANION_CLASS[.@i] != 1637)
			continue;
		if (!#COMPANION_GID[.@i] || !unitexists(#COMPANION_GID[.@i]))
			continue;
		set .@priest, 1;
		break;
	}
	if (!.@priest)
		end;
	if (#COMPANION_RES_TICK > gettimetick(2)) {
		dispbottom "Your priest companion needs more time to recover its divine power...";
		end;
	}
	set #COMPANION_RES_TICK, gettimetick(2) + 60;
	recovery 0, getcharid(0), 4;
	specialeffect2 EF_RESURRECTION;
	dispbottom "Your priest companion has resurrected you!";
	end;
}
